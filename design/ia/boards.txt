class Component extends DCLogic {
  constructor(props) { super(props); this.state = { cur: { L: 4, H: 4, W: 1, V: 2 }, draft: { L: 6, H: 4, W: 2, V: 2 }, phase: 'edit', ver: 4 }; }
  renderVals() {
    const st = this.state, cur = st.cur, dr = st.draft;
    const FF = [512, 1024, 2048], VS = [256, 8192, 50257];
    const cfg = (s) => ({ V: VS[s.V], d: 256, T: 256, L: s.L, H: s.H, ff: FF[s.W] });
    const P = (c) => c.V * c.d + c.T * c.d + c.L * (4 * c.d * c.d + 4 * c.d + 2 * c.d * c.ff + c.ff + c.d + 4 * c.d) + 2 * c.d;
    const C = (c) => 6 * (c.L * (4 * c.d * c.d + 2 * c.d * c.ff) + c.V * c.d) + 6 * c.L * c.T * c.d;
    const TM = (c) => 16 * P(c);
    const MEM = (c) => 4 * P(c) + 2 * c.L * c.d * 4 * c.T;
    const a = cfg(cur), b = cfg(dr);
    const editing = st.phase === 'edit';
    const set = (patch) => { if (editing) this.setState({ draft: Object.assign({}, dr, patch) }); };
    const words = ['Letters', 'Small word list', 'Large word list'], widths = ['Narrow', 'Normal', 'Wide'];
    const changes = [];
    if (dr.L !== cur.L) changes.push((dr.L > cur.L ? 'more' : 'fewer') + ' blocks (' + cur.L + ' → ' + dr.L + ')');
    if (dr.H !== cur.H) changes.push('look-back heads ' + cur.H + ' → ' + dr.H);
    if (dr.W !== cur.W) changes.push(widths[dr.W].toLowerCase() + ' thinking room');
    if (dr.V !== cur.V) changes.push(words[dr.V].toLowerCase());
    const n = st.ver, nv = 'version ' + (n + 1);
    // tower
    const maxL = Math.max(cur.L, dr.L), slabs = [];
    for (let i = 0; i < maxL; i++) {
      const added = i >= cur.L, removed = i >= dr.L;
      slabs.push({ label: 'Block ' + (i + 1), eyes: Array.from({ length: removed ? cur.H : dr.H }, (_, k) => k), thinkW: [36, 72, 144][removed ? cur.W : dr.W] + 'px',
        bg: added ? '#211c12' : '#1b1f27', bd: added ? '1px dashed #e5b567' : '1px solid #2a2f3a', op: removed ? '0.35' : '1',
        tag: added ? 'new' : removed ? 'removed' : (dr.H !== cur.H || dr.W !== cur.W ? 'changed' : ''), tagColor: '#e5b567' });
    }
    const seg = (on, wasCur) => ({ bg: on ? '#9f8cff' : '#1b1f27', fg: on ? '#0e1014' : '#e6e8ec', bd: wasCur && !on ? '1px dashed #9f8cff' : '1px solid #2a2f3a' });
    const controls = [
      { name: 'Look-back heads', hint: 'how many things each block looks for at once', color: '#e58fb0', opts: [1, 4, 8].map((h) => Object.assign({ label: String(h), on: dr.H === h, pick: () => set({ H: h }) }, seg(dr.H === h, cur.H === h))) },
      { name: 'Thinking room', hint: 'space to work on each word', color: '#6cb4ee', opts: widths.map((w, i) => Object.assign({ label: w, on: dr.W === i, pick: () => set({ W: i }) }, seg(dr.W === i, cur.W === i))) }
    ];
    const blocks = { n: dr.L, color: dr.L !== cur.L ? '#e5b567' : '#e6e8ec', was: dr.L !== cur.L ? 'was ' + cur.L : '', less: () => set({ L: Math.max(1, dr.L - 1) }), more: () => set({ L: Math.min(12, dr.L + 1) }) };
    const prompt = 'The largest planet in our solar system is';
    const chips = dr.V === 0 ? prompt.split('').map((c) => (c === ' ' ? '␣' : c)) : dr.V === 2 ? ['The', ' largest', ' planet', ' in', ' our', ' solar', ' system', ' is'] : ['pieces depend on your data'];
    const wordsV = { count: dr.V === 0 ? '41 pieces, one per character' : dr.V === 2 ? '8 pieces' : 'built from your data when you save', chips,
      opts: words.map((w, i) => Object.assign({ label: w, on: dr.V === i, pick: () => set({ V: i }) }, seg(dr.V === i, cur.V === i))) };
    const say = (x, y, more, less) => { const r = y / x; return Math.abs(r - 1) < 0.05 ? 'about the same' : r > 1 ? 'about ' + r.toFixed(1) + '× ' + more : 'about ' + (1 / r).toFixed(1) + '× ' + less; };
    const bar = (x, y) => { const m = Math.max(x, y); return [Math.round(x / m * 100) + '%', Math.round(y / m * 100) + '%']; };
    const costs = [['Size', P(a), P(b), 'bigger', 'smaller'], ['Time per training step', C(a), C(b), 'slower', 'faster'], ['Memory while training', TM(a), TM(b), 'more', 'less']].map(([name, x, y, mo, le]) => {
      const [wa, wb] = bar(x, y), s = say(x, y, mo, le);
      return { name, say: s, a: wa, b: wb, color: s === 'about the same' ? '#8f97a6' : '#e5b567', barB: s === 'about the same' ? '#9f8cff' : '#e5b567' };
    });
    const expect = [];
    if (dr.L > cur.L) expect.push('More blocks: each word gets more rounds of processing, and every training step takes longer. With little data it may memorize rather than learn; watch the test line.');
    if (dr.L < cur.L) expect.push('Fewer blocks: smaller and faster. It may not get its error as low.');
    if (dr.L !== cur.L) expect.push('Last time blocks changed (2 → 4, version 2 → 3), error moved by [N] bits per character.');
    if (dr.H !== cur.H) expect.push('Different look-back heads: same size, same speed. Each block splits its looking into ' + (dr.H > cur.H ? 'more, narrower' : 'fewer, wider') + ' views. Whether that helps at this size is something to test, not assume.');
    if (dr.W > cur.W) expect.push('More thinking room: more to learn in every block, so bigger and slower.');
    if (dr.W < cur.W) expect.push('Less thinking room: smaller and faster, with less capacity in every block.');
    if (dr.V === 0 && cur.V !== 0) expect.push('Letters: the word table becomes tiny, but the model has to learn spelling, and your text turns into about 5 times as many pieces, so each step covers less text.');
    if (dr.V === 1 && cur.V !== 1) expect.push('Small word list: built from your own data. A much smaller table; text splits into somewhat more pieces.');
    if (dr.V === 2 && cur.V !== 2) expect.push('Large word list: GPT-2’s list. A big table, short sequences.');
    if (!expect.length) expect.push('Change something in the picture to see what to expect.');
    // states
    const ph = st.phase;
    const wordChanged = dr.V !== cur.V;
    const startNote = wordChanged ? 'The word list changed, so the old word table doesn’t fit: ' + nv + ' has to start from scratch.'
      : dr.H !== cur.H ? nv[0].toUpperCase() + nv.slice(1) + ' reuses version ' + n + '’s weights split into a different number of heads, so its error jumps at first and then recovers.'
      : (dr.L < cur.L || dr.W < cur.W) ? 'Removed parts are dropped from version ' + n + '’s weights; error rises a little, then recovers.'
      : 'New parts start at zero, so ' + nv + ' begins exactly where version ' + n + ' is and catches up from there.';
    const tray = ph === 'edit' ? (changes.length ? {
        kicker: 'NOT SAVED · ONLY THIS PAGE SEES THESE', chips: changes, text: 'Nothing else in the app sees these yet. Saving makes ' + nv + '.', bg: '#211c12', line: '#e5b567',
        hasQuick: true, hasA: true, aLabel: 'Save as ' + nv, a: () => this.setState({ phase: 'saved' }), hasB: true, bLabel: 'Discard', b: () => this.setState({ draft: Object.assign({}, cur) }) }
      : { kicker: 'NO CHANGES', chips: [], text: 'Version ' + n + ' is the current design. Chat answers with the best checkpoint on held-back text.', bg: '#15181e', line: '#2a2f3a', hasA: false, hasB: false })
      : ph === 'saved' ? { kicker: nv.toUpperCase() + (wordChanged ? ' SAVED · NEEDS TRAINING FROM SCRATCH' : ' SAVED · NEEDS CATCH-UP TRAINING'), chips: [], text: startNote + ' Chat keeps version ' + n + ' until ' + nv + ' beats it on held-back text, or you choose it.', bg: '#26171a', line: '#ef8a7a',
        hasA: true, aLabel: wordChanged ? 'Add training to the queue' : 'Add catch-up to the queue', a: () => this.setState({ phase: 'queued' }), hasB: !wordChanged, bLabel: 'Start from scratch instead', b: () => this.setState({ phase: 'queued' }) }
      : ph === 'queued' ? { kicker: 'QUEUED ON THIS COMPUTER', chips: [], text: nv[0].toUpperCase() + nv.slice(1) + ' waits behind the current job. The sidebar and “Queue and schedule” show where it is.', bg: '#15181e', line: '#6cb4ee', hasA: true, aLabel: 'Move it to the front', a: () => this.setState({ phase: 'training' }), hasB: false }
      : { kicker: nv.toUpperCase() + (wordChanged ? ' IS TRAINING' : ' IS CATCHING UP'), chips: [], text: 'Version ' + n + '’s run is paused and can resume. Chat switches to ' + nv + ' only when it beats version ' + n + ' on held-back text, or when you pick it.', bg: '#1a1730', line: '#9f8cff', hasA: true, aLabel: 'Start new changes', a: () => this.setState({ phase: 'edit', ver: n + 1, cur: Object.assign({}, dr), draft: Object.assign({}, dr) }), hasB: false };
    const run = ph === 'training' ? { pill: nv[0].toUpperCase() + nv.slice(1) + (wordChanged ? ' is training' : ' is catching up'), dot: '#9f8cff', w: '4%' } : { pill: 'Version ' + n + ' is training', dot: '#9f8cff', w: '62%' };
    const see = ph === 'training' ? { v5color: '#e5b567', trainV5: '- - ' + nv + ': ' + (wordChanged ? 'starts from scratch' : 'starts from version ' + n + '’s level'), answerV5: 'answers from its best checkpoint so far', path: wordChanged ? 'M30 22 C 90 80, 150 112, 240 128' : 'M470 142 C 500 140, 540 146, 590 149' }
      : { v5color: '#5b6271', trainV5: nv + ': appears once it trains', answerV5: ph === 'edit' ? 'save your changes and train them to compare' : 'has no trained weights yet', path: '' };
    const machine = ph === 'training' ? { now: nv, what: wordChanged ? 'from scratch' : 'catching up', pct: 4, w: '4%', next: 'about [N] min left · version ' + n + ' paused' }
      : ph === 'queued' ? { now: 'version ' + n, what: 'steady phase', pct: 62, w: '62%', next: 'about [N] min left · then ' + nv }
      : { now: 'version ' + n, what: 'steady phase', pct: 62, w: '62%', next: 'about [N] min left · nothing queued' };
    return { slabs, controls, blocks, words: wordsV, costs, expect, tray, run, see, machine };
  }
}

class Component extends DCLogic {
  constructor(props) { super(props); this.state = { use: 3, b: 2, compare: false, how: false }; }
  renderVals() {
    const st = this.state;
    const BEST = 3;
    const V = [
      { name: 'Version 1', change: 'Starting design: GPT-2 style', chip: 'baseline', tone: 'n', answers: 'not yet' },
      { name: 'Version 2', change: 'Rotating positions instead of learned ones', chip: 'better', tone: 'g', answers: 'not yet' },
      { name: 'Version 3', change: '4 blocks instead of 2', chip: 'better', tone: 'g', answers: 'not yet' },
      { name: 'Version 4', change: 'Simpler normalization (RMSNorm)', chip: 'within the spread', tone: 'n', answers: 'yes' },
      { name: 'Version 5', change: '6 blocks and wide thinking room', chip: 'catching up', tone: 'c', answers: 'not yet', training: true }
    ];
    const col = { g: '#9f8cff', n: '#8f97a6', c: '#ef8a7a' };
    const A = '#9f8cff', B = '#6cb4ee';
    const versions = V.map((v, i) => {
      const isA = i === st.use, isB = st.compare && i === st.b;
      const on = isA || isB;
      return { name: v.name, change: v.change, chip: v.chip, chipColor: col[v.tone], blocked: false, op: '1', on,
        open: isA && !st.compare,
        bd: isA ? '2px solid ' + A : isB ? '2px solid ' + B : '1px solid #2a2f3a', bg: on ? '#1b1f27' : 'transparent',
        slot: isA ? (st.compare ? 'left reply' : 'answering now') : isB ? 'right reply' : i === BEST ? 'best on held-back text' : v.training ? 'still training · answers from its best checkpoint so far' : '',
        slotColor: isA ? A : isB ? B : '#8f97a6',
        pick: () => { if (st.compare) { if (i !== st.use) this.setState({ b: i }); } else this.setState({ use: i, b: i > 0 ? i - 1 : 1 }); } };
    });
    const a = V[st.use], b = V[st.b];
    const older = st.use < st.b ? a : b, newer = st.use < st.b ? b : a;
    const unfinished = a.training || b.training;
    const cur = V[st.use];
    return {
      versions, compare: st.compare, single: !st.compare, how: st.how,
      head: { using: cur.name + ' · checkpoint [N]', mode: st.use === BEST ? 'follows the best checkpoint' : 'picked by you · ' + (V[BEST].name) + ' is best' },
      a: { tag: cur.name + ' · checkpoint [N]' + (cur.answers === 'yes' ? '' : ' · hasn’t practised answering yet') },
      pair: [{ tag: a.name, color: A }, { tag: b.name, color: B }],
      cmpBtn: st.compare ? { label: 'Back to one version', bg: '#9f8cff', fg: '#0e1014' } : { label: 'Compare two versions', bg: 'transparent', fg: '#9f8cff' },
      toggleCompare: () => this.setState({ compare: !st.compare, how: false }),
      toggleHow: () => this.setState({ how: !st.how }), howLabel: st.how ? 'Hide how this is measured' : 'How this is measured',
      verdict: unfinished
        ? { color: '#e5b567', line: 'No verdict yet: version 5 hasn’t finished its schedule.', sub: 'Error mid-schedule looks worse than it will after the cool-down, so only runs trained to the same length on the same schedule are compared. Its replies are still worth reading.' }
        : { color: '#e6e8ec', line: 'To reach the same error, ' + newer.name + ' needs [N]% of the training ' + older.name + ' needed.', sub: 'Under 100% means the change saves training. Held at [N] of 3 sizes · larger than the spread across 3 seeds · same length and schedule · error per character on held-back text' },
      score: [{ name: 'Error on held-back text', val: '[N] bits/char' }, { name: 'Answers prompts', val: cur.answers }, { name: 'Checked problems passed', val: '[k] of [n]' }, { name: 'Judge preferred it over previous', val: '[k] of [n]' }, { name: 'Size · speed', val: '[N]M · [N] pieces/s' }],
      shelfHint: st.compare ? 'Pick the version for the right-hand reply.' : 'Chat follows the best checkpoint unless you pick one. Switching is live; nothing retrains.',
      placeholder: st.compare ? 'Ask both versions the same thing' : 'Type a message'
    };
  }
}

class Component extends DCLogic {
  constructor(props) { super(props); this.state = { mode: 'params', lr: true, guess: -1 }; }
  renderVals() {
    const st = this.state, mode = st.mode;
    const M = {
      params: { name: 'Parameters', note: 'Each variant’s width is re-solved so its parameter count lands within about 2% of the base. Same data and amount of text for all.', x: 'TEXT SEEN' },
      compute: { name: 'Compute', note: 'Each run stops at about the same total training compute. Cheaper variants see more text.', x: 'TRAINING COMPUTE' },
      tokens: { name: 'Text seen', note: 'Every run sees the same text. Sizes are allowed to differ, so read the comparison with that in mind.', x: 'TEXT SEEN' },
      time: { name: 'Wall-clock', note: 'Every run gets the same minutes on this computer. Rewards variants that are fast here; may not hold on other hardware.', x: 'MINUTES' }
    };
    const adj = {
      params: ['—', 'width re-solved · [N]M', 'none needed · [N]M', 'none needed · [N]M'],
      compute: ['—', 'stops at base compute · sees about [N]% more text', 'same compute', 'same compute'],
      tokens: ['—', '[N]M parameters', '[N]M parameters', '[N]M parameters'],
      time: ['—', '[N] min budget', '[N] min budget', '[N] min budget']
    }[mode];
    const base = [
      { id: 'A', title: 'Base: version 4', diff: 'base spec', color: '#e6e8ec', delta: '—' },
      { id: 'B', title: 'Mamba-style mixer', diff: 'mixer: attention → ssm · channel: mlp_gelu → none · position: rope → none', color: '#9f8cff', delta: '[N] · inside the spread?' },
      { id: 'C', title: 'Wide thinking room', diff: 'ff: 1024 → 2048', color: '#e5b567', delta: '[N]' },
      { id: 'D', title: 'Muon optimizer', diff: 'optimizer: adamw → muon', color: '#6cb4ee', delta: '[N]' }
    ];
    const runs = base.map((r, i) => Object.assign({ adjust: i === 0 ? adj[i] : adj[i] + (st.lr ? '' : ' · learning rate not retuned') }, r));
    const modes = Object.keys(M).map((k) => ({ name: M[k].name, on: mode === k, bg: mode === k ? '#9f8cff' : '#1b1f27', fg: mode === k ? '#0e1014' : '#e6e8ec', pick: () => this.setState({ mode: k }) }));
    const guess = ['Better', 'About the same', 'Worse'].map((label, i) => ({ label, on: st.guess === i, bg: st.guess === i ? '#e58fb0' : '#1b1f27', fg: st.guess === i ? '#0e1014' : '#e6e8ec', pick: () => this.setState({ guess: i }) }));
    return {
      modes, runs, guess, modeNote: M[mode].note, xAxis: M[mode].x,
      lr: { on: st.lr, track: st.lr ? '#9f8cff' : '#2a2f3a', knob: st.lr ? '18px' : '2px',
        note: st.lr ? 'Each variant first gets a short learning-rate sweep, so none loses just because the base’s rate suits it badly.' : 'Variants use the base’s learning rate; a variant can lose only because that rate suits it badly.' },
      toggleLr: () => this.setState({ lr: !st.lr }),
      runLabel: st.lr ? 'Queue 12 runs + 9 short sweeps' : 'Queue 12 runs'
    };
  }
}

    const curve = (k, floor, upto) => { let d = ''; for (let t = 0; t <= upto; t += 2) d += (t ? ' L ' : 'M ') + x(t) + ' ' + y(floor + (10.82 - floor) * Math.exp(-t / k)); return d; };
    const guess = ['Lower', 'Same', 'Higher'].map((label, i) => Object.assign({ label, on: st.guess === i, pick: () => this.setState({ guess: i }) }, { bg: st.guess === i ? '#e58fb0' : '#1b1f27', fg: st.guess === i ? '#0e1014' : '#e6e8ec' }));
    const plot = { uniY: y(7.4), uniTy: (y(7.4) - 5), a: running || done ? curve(16, 4.4, 100) : '', b: done ? curve(19, 4.2, 100) : running ? curve(19, 4.2, 45) : '', refY: y(10.82), refTy: (y(10.82) - 5) };
    const st4 = done ? ['✓', '✓', '✓', '✓'] : running ? ['✓', '✓', '…', '…'] : ['', '', '', ''];
    const rings = st4.map((m) => (m === '✓' ? '#9f8cff' : '#5b6271'));
    const checks = [
      ['Starts where it should', 'First error near the uniform-guess level.'],
      ['Numbers stay finite', 'No overflow or not-a-number in errors or gradients.'],
      ['Can memorize one batch', 'Drives its error on one small batch close to zero.'],
      ['Error falls', 'Smoothed error drops by more than the step-to-step wobble.']
    ].map(([name, what], i) => ({ name, what, mark: st4[i], ring: rings[i] }));
    const phases = [['setup', 'Ready'], ['running', 'Running'], ['done', 'Done']].map(([id, label]) => ({ label, on: ph === id, bg: ph === id ? '#2a2f3a' : 'transparent', fg: ph === id ? '#e6e8ec' : '#8f97a6', pick: () => this.setState({ phase: id }) }));
    return {
      choices, bars, plot, guess, guessed: ['lower', 'about the same', 'higher'][st.guess] || 'nothing', checks, phases, running, done, hasResults: ph !== 'setup', how: st.how,
      res: { kicker: done ? 'RESULT · SAME DATA, ORDER AND TIME' : 'RUNNING · ONE AFTER THE OTHER, SAME DATA ORDER' },
      run: () => this.setState({ phase: 'running' }), runLabel: running ? 'Running…' : done ? 'Run again' : 'Run quick test',
      toggleHow: () => this.setState({ how: !st.how }), howLabel: st.how ? 'Hide how quick tests work' : 'How quick tests work'
    };
  }
}
